package com.example.map.model;

import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;

@Entity
public class Location {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    private String name;
    private String description;
    private Double lat;
    private Double lng;

    public Location() {}

    public Location(String name, String description, Double lat, Double lng) {
        this.name = name;
        this.description = description;
        this.lat = lat;
        this.lng = lng;
    }

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }
    public String getName() { return name; }
    public void setName(String name) { this.name = name; }
    public String getDescription() { return description; }
    public void setDescription(String description) { this.description = description; }
    public Double getLat() { return lat; }
    public void setLat(Double lat) { this.lat = lat; }
    public Double getLng() { return lng; }
    public void setLng(Double lng) { this.lng = lng; }
}

package com.example.map.service;

import com.example.map.model.Location;
import com.example.map.repository.LocationRepository;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.web.server.ResponseStatusException;

import java.util.List;

@Service
public class LocationService {
    private final LocationRepository repository;

    public LocationService(LocationRepository repository) {
        this.repository = repository;
    }

    public Location create(Location location) {
        validate(location);
        return repository.save(location);
    }

    public List<Location> findAll() {
        return repository.findAll();
    }

    public Location findById(Long id) {
        return repository.findById(id)
            .orElseThrow(() -> notFound(id));
    }

    public Location update(Long id, Location changes) {
        Location existing = findById(id);
        validate(changes);
        existing.setName(changes.getName());
        existing.setDescription(changes.getDescription());
        existing.setLat(changes.getLat());
        existing.setLng(changes.getLng());
        return repository.save(existing);
    }

    public void delete(Long id) {
        if (!repository.existsById(id)) {
            throw notFound(id);
        }
        repository.deleteById(id);
    }

    private void validate(Location l) {
        if (l == null || l.getName() == null || l.getName().isBlank()) {
            throw badRequest("name is required");
        }
        if (l.getName().length() > 120) {
            throw badRequest("name must be at most 120 characters");
        }
        if (l.getDescription() != null && l.getDescription().length() > 500) {
            throw badRequest("description must be at most 500 characters");
        }
        if (l.getLat() != null && (l.getLat() < -90 || l.getLat() > 90)) {
            throw badRequest("lat must be within [-90,90]");
        }
        if (l.getLng() != null && (l.getLng() < -180 || l.getLng() > 180)) {
            throw badRequest("lng must be within [-180,180]");
        }
    }

    private ResponseStatusException notFound(Long id) {
        return new ResponseStatusException(HttpStatus.NOT_FOUND,
            "location " + id + " not found");
    }

    private ResponseStatusException badRequest(String msg) {
        return new ResponseStatusException(HttpStatus.BAD_REQUEST, msg);
    }
}

package com.example.map.controller;

import com.example.map.model.Location;
import com.example.map.service.LocationService;
import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/locations")
public class LocationController {
    private final LocationService service;

    public LocationController(LocationService service) {
        this.service = service;
    }

    @PostMapping
    @ResponseStatus(HttpStatus.CREATED)
    public Location create(@RequestBody Location location) {
        return service.create(location);
    }

    @GetMapping
    public List<Location> list() {
        return service.findAll();
    }

    @GetMapping("/{id}")
    public Location get(@PathVariable Long id) {
        return service.findById(id);
    }

    @PutMapping("/{id}")
    public Location update(@PathVariable Long id, @RequestBody Location changes) {
        return service.update(id, changes);
    }

    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    public void delete(@PathVariable Long id) {
        service.delete(id);
    }
}

package com.example.map;

import com.example.map.model.Location;
import com.example.map.repository.LocationRepository;
import com.example.map.service.LocationService;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;
import org.springframework.web.server.ResponseStatusException;

import java.util.Optional;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.when;

class LocationServiceTest {
    private LocationRepository repo;
    private LocationService service;

    @BeforeEach
    void setUp() {
        repo = Mockito.mock(LocationRepository.class);
        service = new LocationService(repo);
    }

    @Test
    void createReturnsSavedLocationWithGeneratedId() {
        when(repo.save(any())).thenAnswer(inv -> {
            Location l = inv.getArgument(0);
            l.setId(1L);
            return l;
        });
        Location saved = service.create(new Location("Office", "HQ", null, null));
        assertEquals(1L, saved.getId());
        assertEquals("Office", saved.getName());
    }

    @Test
    void createRejectsBlankName() {
        assertThrows(ResponseStatusException.class,
            () -> service.create(new Location(" ", "x", null, null)));
    }

    @Test
    void createRejectsOversizedName() {
        assertThrows(ResponseStatusException.class,
            () -> service.create(new Location("n".repeat(121), "x", null, null)));
    }

    @Test
    void createRejectsOutOfRangeCoordinates() {
        assertThrows(ResponseStatusException.class,
            () -> service.create(new Location("ok", "x", 95.0, null)));
        assertThrows(ResponseStatusException.class,
            () -> service.create(new Location("ok", "x", null, -181.0)));
    }

    @Test
    void findByIdUnknownThrows404() {
        when(repo.findById(99L)).thenReturn(Optional.empty());
        ResponseStatusException ex = assertThrows(ResponseStatusException.class,
            () -> service.findById(99L));
        assertEquals(404, ex.getStatusCode().value());
    }

    @Test
    void updateChangesFields() {
        Location existing = new Location("old", "d", null, null);
        existing.setId(5L);
        when(repo.findById(5L)).thenReturn(Optional.of(existing));
        when(repo.save(any())).thenAnswer(inv -> inv.getArgument(0));
        Location updated = service.update(5L, new Location("new", "d2", 1.0, 2.0));
        assertEquals("new", updated.getName());
        assertEquals(1.0, updated.getLat());
    }

    @Test
    void deleteUnknownThrows404() {
        when(repo.existsById(7L)).thenReturn(false);
        ResponseStatusException ex = assertThrows(ResponseStatusException.class,
            () -> service.delete(7L));
        assertEquals(404, ex.getStatusCode().value());
    }
}

package com.example.map;

import com.example.map.controller.LocationController;
import com.example.map.model.Location;
import com.example.map.service.LocationService;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.servlet.WebMvcTest;
import org.springframework.boot.test.mock.mockito.MockBean;
import org.springframework.test.web.servlet.MockMvc;

import java.util.List;

import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.when;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.*;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.*;

@WebMvcTest(LocationController.class)
class LocationControllerTest {
    @Autowired
    private MockMvc mvc;

    @MockBean
    private LocationService service;

    @Test
    void postCreatesAndReturns201() throws Exception {
        when(service.create(any())).thenAnswer(inv -> {
            Location l = inv.getArgument(0);
            l.setId(1L);
            return l;
        });
        mvc.perform(post("/api/locations")
                .contentType("application/json")
                .content("{\"name\":\"Office\",\"description\":\"HQ\"}"))
            .andExpect(status().isCreated())
            .andExpect(jsonPath("$.id").value(1))
            .andExpect(jsonPath("$.name").value("Office"));
    }

    @Test
    void getListReturnsAll() throws Exception {
        when(service.findAll()).thenReturn(List.of(
            new Location("A", "a", 1.0, 2.0)));
        mvc.perform(get("/api/locations"))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$[0].name").value("A"));
    }

    @Test
    void getUnknownReturns404() throws Exception {
        when(service.findById(99L)).thenThrow(new ResponseStatusExceptionHolder().ex());
        mvc.perform(get("/api/locations/99")).andExpect(status().isNotFound());
    }

    @Test
    void putUpdatesReturns200() throws Exception {
        when(service.update(Mockito.eq(3L), any()))
            .thenReturn(new Location("upd", "d", null, null));
        mvc.perform(put("/api/locations/3")
                .contentType("application/json")
                .content("{\"name\":\"upd\",\"description\":\"d\"}"))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.name").value("upd"));
    }

    static class ResponseStatusExceptionHolder {
        org.springframework.web.server.ResponseStatusException ex =
            new org.springframework.web.server.ResponseStatusException(
                org.springframework.http.HttpStatus.NOT_FOUND, "nope");
        org.springframework.web.server.ResponseStatusException ex() { return ex; }
    }
}
