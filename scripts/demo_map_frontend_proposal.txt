import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { HttpClient } from '@angular/common/http';

interface LocationDto {
  id?: number;
  name: string;
  description?: string;
  lat?: number;
  lng?: number;
}

@Component({
  selector: 'app-location-list',
  standalone: true,
  imports: [CommonModule, FormsModule],
  template: `
    <h1>Locations</h1>
    <form (ngSubmit)="add()">
      <div><input placeholder="Name" [(ngModel)]="name" name="name" required></div>
      <div><textarea placeholder="Description" [(ngModel)]="description"
             name="description"></textarea></div>
      <div><input placeholder="lat (optional)" [(ngModel)]="lat" name="lat">
           <input placeholder="lng (optional)" [(ngModel)]="lng" name="lng"></div>
      <button type="submit">Add location</button>
    </form>
    <p *ngIf="message">{{ message }}</p>
    <table>
      <thead><tr><th>#</th><th>Name</th><th>Description</th><th>Lat</th><th>Lng</th></tr></thead>
      <tbody>
        <tr *ngFor="let l of locations">
          <td>{{ l.id }}</td><td>{{ l.name }}</td><td>{{ l.description }}</td>
          <td>{{ l.lat }}</td><td>{{ l.lng }}</td>
        </tr>
      </tbody>
    </table>
  `
})
export class LocationListComponent implements OnInit {
  locations: LocationDto[] = [];
  name = '';
  description = '';
  lat = '';
  lng = '';
  message = '';

  constructor(private http: HttpClient) {}

  ngOnInit(): void {
    this.load();
  }

  load(): void {
    this.http.get<LocationDto[]>('/api/locations')
      .subscribe(data => this.locations = data);
  }

  add(): void {
    const payload: LocationDto = {
      name: this.name,
      description: this.description
    };
    if (this.lat) payload.lat = Number(this.lat);
    if (this.lng) payload.lng = Number(this.lng);
    this.http.post<LocationDto>('/api/locations', payload)
      .subscribe({
        next: () => {
          this.message = 'Created.';
          this.name = ''; this.description = ''; this.lat = ''; this.lng = '';
          this.load();
        },
        error: () => this.message = 'Create failed.'
      });
  }
}
